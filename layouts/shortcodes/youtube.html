{{- /* YouTube video or Short, loaded only on click (no Google requests until played). Usage: {{< youtube id="rEL1o0tGs_M" caption="..." credit="..." >}} id accepts a bare ID or any youtube.com / youtu.be / shorts URL. ratio: "vertical" (default for /shorts/ URLs) or "wide". title: accessible label. */ -}} {{- $in := .Get "id" | default (.Get 0) | default "" -}} {{- $id := $in -}} {{- with findRESubmatch `(?:shorts/|v=|youtu\.be/|embed/)([A-Za-z0-9_-]{11})` $in 1 }}{{ $id = index (index . 0) 1 }}{{ end -}} {{- if not (findRE `^[A-Za-z0-9_-]{11}$` $id) }}{{ errorf "youtube shortcode: bad id %q" $in }}{{ end -}} {{- $ratio := .Get "ratio" | default (cond (strings.Contains $in "/shorts/") "vertical" "wide") -}} {{- $title := .Get "title" | default (.Get "caption") | default "video" -}} {{- /* Poster is fetched once at build time and served from this site. */ -}} {{- $poster := "" -}} {{- with .Get "poster" }}{{ with resources.Get (printf "img/%s" .) }}{{ $poster = . }}{{ end }} {{- else }} {{- $name := cond (eq $ratio "vertical") "hq2" "hqdefault" -}} {{- with try (resources.GetRemote (printf "https://i.ytimg.com/vi/%s/%s.jpg" $id $name)) }}{{ with .Value }}{{ $poster = . }}{{ else }}{{ warnf "youtube poster not fetched for %s: %v" $id .Err }}{{ end }}{{ end }} {{- end -}}
